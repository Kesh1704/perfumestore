${p.name}

$${p.price}

`; container.appendChild(div); }); } function addToCart(id, name, price) { const cart = JSON.parse(localStorage.getItem('cart')) || []; const index = cart.findIndex(item => item.id === id); if (index !== -1) { cart[index].qty += 1; } else { cart.push({ id, name, price, qty: 1 }); } localStorage.setItem('cart', JSON.stringify(cart)); alert(`${name} added to cart!`); } loadProducts();
